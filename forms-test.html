Query 1 — Results total (last 30 days)

SELECT count(*) AS results_total
  FROM public.form_submissions s
  JOIN public.forms f ON f.id = s.form_id
 WHERE f.form_number = 'FRM-000001'
   AND f.name = 'Connect with us'
   AND s.is_spam = false
   AND s.submitted_at >= now() - interval '30 days';

Query 2 — Latest submissions and how they were sourced

SELECT s.submitted_at,
       s.raw_data->>'email'            AS email,
       fs.surface_type,
       fs.name                         AS surface_name,
       s.landing_page_url,
       lp.name                         AS landing_page,
       s.delivery_status,
       s.delivery_error
  FROM public.form_submissions s
  JOIN public.forms f               ON f.id  = s.form_id
  LEFT JOIN public.form_surfaces fs ON fs.id = s.form_surface_id
  LEFT JOIN public.landing_pages lp ON lp.id = s.landing_page_id
 WHERE f.form_number = 'FRM-000001'
   AND f.name = 'Connect with us'
 ORDER BY s.submitted_at DESC
 LIMIT 15;
