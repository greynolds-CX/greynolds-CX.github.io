Query 1 — this test's submissions (last 3 hours)

SELECT s.submitted_at,
       s.raw_data->>'name'   AS name,
       s.raw_data->>'email'  AS email,
       s.landing_page_url,
       s.delivery_status,
       s.delivery_error
  FROM public.form_submissions s
  JOIN public.forms f ON f.id = s.form_id
 WHERE f.form_number = 'FRM-000001'
   AND f.name = 'Connect with us'
   AND s.submitted_at >= now() - interval '3 hours'
 ORDER BY s.submitted_at DESC;
